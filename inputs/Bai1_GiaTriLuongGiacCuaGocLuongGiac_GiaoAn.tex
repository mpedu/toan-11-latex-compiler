\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Năng lực Toán học đặc thù}:
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Năng lực tư duy và lập luận toán học}:
        \begin{itemize}
            \item Nhận biết được các khái niệm cơ bản về góc lượng giác: khái niệm góc lượng giác, số đo của góc lượng giác (đơn vị độ và radian), hệ thức Chasles cho các góc lượng giác, đường tròn lượng giác.
            \item Nhận biết được khái niệm giá trị lượng giác của một góc lượng giác ($\sin\alpha$, $\cos\alpha$, $\tan\alpha$, $\cot\alpha$).
            \item Giải thích được dấu của các giá trị lượng giác dựa trên góc phần tư chứa điểm biểu diễn của góc lượng giác trên đường tròn lượng giác.
            \item Mô tả và ghi nhớ bảng giá trị lượng giác của một số góc lượng giác thường gặp; thiết lập và vận dụng các hệ thức cơ bản giữa các giá trị lượng giác của một góc lượng giác: $\sin^2\alpha + \cos^2\alpha = 1$, $1 + \tan^2\alpha = \dfrac{1}{\cos^2\alpha}$, $1 + \cot^2\alpha = \dfrac{1}{\sin^2\alpha}$, $\tan\alpha \cdot \cot\alpha = 1$.
            \item Nhận biết và vận dụng mối quan hệ giữa các giá trị lượng giác của các góc lượng giác có liên quan đặc biệt: đối nhau, bù nhau, phụ nhau, hơn kém nhau $\pi$.
        \end{itemize}
        \item \textit{Năng lực mô hình hóa toán học}: Thiết lập mô hình lượng giác để mô tả các hiện tượng chuyển động tròn đều trong thực tiễn (chuyển động của kim đồng hồ, đu quay, ròng rọc cơ học, bánh xe đạp).

        \item \textit{Năng lực giao tiếp toán học}: Sử dụng chính xác các thuật ngữ, ký hiệu toán học lượng giác; trình bày lời giải rõ ràng, mạch lạc; tích cực trao đổi, phản biện trong hoạt động nhóm.
        \item \textit{Năng lực sử dụng công cụ, phương tiện học toán}: Sử dụng thành thạo máy tính cầm tay (MTCT Casio fx-580VN X) để chuyển đổi giữa đơn vị độ và radian, tính giá trị lượng giác của một góc và xác định số đo góc khi biết giá trị lượng giác; khai thác phần mềm trực quan GeoGebra để khảo sát đường tròn lượng giác.
    \end{itemize}

\subsubsection*{1. Hoạt động 1: Mở đầu (Khởi động - 7 phút)}

\paragraph{a) Mục tiêu:}
Tạo tình huống có vấn đề xuất phát từ chuyển động quay thực tiễn (chuyển động của kim đồng hồ và vòng quay đu quay), giúp học sinh nhận thấy góc hình học ở lớp 10 (chỉ từ $0^\circ$ đến $180^\circ$) không đủ để mô tả chiều quay và các vòng quay lặp lại nhiều lần, từ đó kích thích nhu cầu mở rộng khái niệm góc lượng giác.

\paragraph{b) Nội dung:}
Giáo viên chiếu video ngắn hoặc hình ảnh chuyển động của kim phút đồng hồ và bánh xe vòng quay đu quay:
\begin{itemize}
    \item \textit{Tình huống 1}: Từ lúc 12 giờ đến 12 giờ 45 phút, kim phút đã quay quanh trục một góc bao nhiêu độ? Nếu kim phút quay tiếp 2 vòng nữa thì tổng số đo góc quay là bao nhiêu? Chiều quay của kim đồng hồ có gì khác biệt so với chiều ngược lại?
    \item \textit{Tình huống 2}: Một chiếc cabin đu quay quay từ vị trí ban đầu qua 3 vòng rồi dừng lại ở vị trí tạo với vị trí ban đầu góc $45^\circ$. Làm thế nào để phân biệt chiều quay thuận, ngược và tổng góc mà cabin đã quét qua?
\end{itemize}

\paragraph{c) Sản phẩm:}
\begin{itemize}
    \item Học sinh nhận diện: Kim phút từ 12 giờ đến 12 giờ 45 phút quay được một góc hình học $270^\circ$.
    \item Khi kim phút quay tiếp 2 vòng nữa, góc quay tích lũy thêm $2 \cdot 360^\circ = 720^\circ$, tổng cộng $270^\circ + 720^\circ = 990^\circ$.
    \item Học sinh phát hiện: Cần có một quy ước về \textbf{chiều quay} (chiều dương, chiều âm) và số đo góc có thể vượt quá $360^\circ$ cũng như có thể mang dấu âm.
\end{itemize}

\paragraph{d) Tổ chức thực hiện:}
\begin{itemize}
    \item \textbf{Bước 1 (Chuyển giao nhiệm vụ)}: GV nêu vấn đề, yêu cầu HS quan sát hình ảnh đồng hồ treo tường và thảo luận cặp đôi trong 2 phút để trả lời hai tình huống trên.
    \item \textbf{Bước 2 (Thực hiện nhiệm vụ - Có giàn giáo)}:
    \begin{itemize}
        \item \textit{Giàn giáo cho HS trung bình - yếu}: GV gợi ý: Một vòng tròn tương ứng bao nhiêu độ? Kim phút đi từ số 12 đến số 9 là mấy phần của vòng tròn? Khi kim tiếp tục quay thì góc quay tăng lên hay bị giới hạn ở $360^\circ$?
    \end{itemize}
    \item \textbf{Bước 3 (Báo cáo, thảo luận)}: Đại diện 1 cặp đôi trả lời, các cặp đôi khác nhận xét, bổ sung.
    \item \textbf{Bước 4 (Kết luận, nhận định)}: GV chốt lại: Trong thực tiễn và kỹ thuật, các chuyển động quay đòi hỏi phải xác định được cả tia đầu, tia cuối, chiều quay và số vòng quay. Đó là lý do ta xây dựng khái niệm \textbf{Góc lượng giác}.
\end{itemize}

\subsubsection*{2. Hoạt động 2: Hình thành kiến thức (23 phút)}

\paragraph{Nội dung 1: Khái niệm góc lượng giác và số đo của góc lượng giác (10 phút)}
\begin{itemize}
    \item \textbf{a) Mục tiêu}: Nhận biết khái niệm góc lượng giác tạo bởi tia đầu $Ou$, tia cuối $Ov$; quy ước chiều quay (chiều dương là ngược chiều kim đồng hồ, chiều âm là cùng chiều kim đồng hồ); kí hiệu $(Ou, Ov)$ và công thức số đo tổng quát của góc lượng giác.
    \item \textbf{b) Nội dung}: Học sinh đọc SGK, quan sát mô phỏng và thực hiện HĐKP 1:
    \begin{itemize}
        \item Trong mặt phẳng, cho hai tia $Ou, Ov$. Xét một tia $Om$ quay quanh điểm $O$ từ tia $Ou$ đến trùng với tia $Ov$.
        \item Quy ước chiều quay: Chiều ngược chiều kim đồng hồ là chiều dương ($+$), chiều cùng chiều kim đồng hồ là chiều âm ($-$).
        \item Khi tia $Om$ quay quanh $O$ từ $Ou$ đến $Ov$, nó tạo nên góc lượng giác có tia đầu $Ou$ và tia cuối $Ov$, kí hiệu là $(Ou, Ov)$.
        \item Số đo của góc lượng giác $(Ou, Ov)$ được xác định bởi: $\text{sđ}(Ou, Ov) = \alpha + k \cdot 360^\circ$ ($k \in \mathbb{Z}$), trong đó $\alpha$ là số đo của một góc lượng giác cụ thể bất kì tạo bởi hai tia đó.
    \end{itemize}
    \item \textbf{c) Sản phẩm}:
    \begin{itemize}
        \item Định nghĩa hoàn chỉnh trong vở ghi.
        \item Ví dụ 1: Cho góc hình học $\widehat{uOv} = 60^\circ$. Nếu tia $Om$ quay theo chiều dương từ $Ou$ đến $Ov$ thì số đo là $60^\circ$. Nếu quay tiếp 1 vòng theo chiều dương thì số đo là $60^\circ + 360^\circ = 420^\circ$. Nếu quay theo chiều âm từ $Ou$ đến $Ov$ thì số đo là $-300^\circ$.
        \item Công thức tổng quát: $\text{sđ}(Ou, Ov) = 60^\circ + k \cdot 360^\circ$ ($k \in \mathbb{Z}$).
    \end{itemize}
    \item \textbf{d) Tổ chức thực hiện}:
    \begin{itemize}
        \item \textit{Chuyển giao}: GV dùng compa hoặc mô hình cánh quạt để biểu diễn trực quan chuyển động quay. Yêu cầu HS viết số đo góc tương ứng với các chiều quay.
        \item \textit{Thực hiện có giàn giáo}: Với HS trung bình - yếu, GV nhấn mạnh: Góc hình học chỉ có 1 giá trị duy nhất từ $0^\circ$ đến $180^\circ$, còn góc lượng giác có vô số số đo sai khác nhau một bội nguyên của $360^\circ$.
        \item \textit{Báo cáo - Nhận định}: Cho 2 HS lên bảng viết biểu thức số đo, GV nhận xét và chuẩn hóa quy ước.
    \end{itemize}
\end{itemize}

\paragraph{Nội dung 2: Đơn vị radian và chuyển đổi giữa độ và radian (7 phút)}
\begin{itemize}
    \item \textbf{a) Mục tiêu}: Nhận biết đơn vị đo góc radian, định nghĩa góc $1\text{ rad}$, nắm vững mối liên hệ $180^\circ = \pi\text{ rad}$ và thành thạo kỹ năng bấm MTCT Casio fx-580VN X chuyển đổi giữa hai đơn vị đo.
    \item \textbf{b) Nội dung}:
    \begin{itemize}
        \item Trên đường tròn bán kính $R$, cung có độ dài bằng bán kính $R$ gọi là cung có số đo $1\text{ rad}$. Góc ở tâm chắn cung đó có số đo là $1\text{ rad}$.
        \item Mối liên hệ: $180^\circ = \pi\text{ rad} \implies 1^\circ = \dfrac{\pi}{180}\text{ rad}$ và $1\text{ rad} = \left(\dfrac{180}{\pi}\right)^\circ \approx 57^\circ 17' 45''$.
        \item Đổi góc $\alpha^\circ$ sang radian: $a = \alpha \cdot \dfrac{\pi}{180}$. Đổi $a\text{ rad}$ sang độ: $\alpha = a \cdot \dfrac{180}{\pi}$.

\paragraph{Nội dung 1: Định nghĩa giá trị lượng giác của góc lượng giác và xét dấu (12 phút)}
\begin{itemize}
    \item \textbf{a) Mục tiêu}: Nhận biết định nghĩa $\cos\alpha, \sin\alpha, \tan\alpha, \cot\alpha$ của góc lượng giác $\alpha$ thông qua tọa độ điểm $M(x; y)$ trên đường tròn lượng giác; hiểu bản chất trục hoành là trục cosin, trục tung là trục sin; xác định dấu của các giá trị lượng giác ở 4 góc phần tư.
    \item \textbf{b) Nội dung}:
    \begin{itemize}
        \item Lấy điểm $M(x; y)$ trên đường tròn lượng giác sao cho $\text{sđ}(OA, OM) = \alpha$. Khi đó:
        \begin{itemize}
            \item Tung độ $y$ của $M$ gọi là sin của $\alpha$, kí hiệu là $\sin\alpha$: $\sin\alpha = y$.
            \item Hoành độ $x$ của $M$ gọi là cosin của $\alpha$, kí hiệu là $\cos\alpha$: $\cos\alpha = x$.
            \item Nếu $\cos\alpha \neq 0$ ($x \neq 0$), tỉ số $\dfrac{\sin\alpha}{\cos\alpha} = \dfrac{y}{x}$ gọi là tang của $\alpha$, kí hiệu là $\tan\alpha$.
            \item Nếu $\sin\alpha \neq 0$ ($y \neq 0$), tỉ số $\dfrac{\cos\alpha}{\sin\alpha} = \dfrac{x}{y}$ gọi là côtang của $\alpha$, kí hiệu là $\cot\alpha$.
        \end{itemize}

        \begin{center}
        \begin{tabular}{|c|c|c|c|c|}
            \hline

            \hline
            $\cos\alpha$ ($x$) & $+$ & $-$ & $-$ & $+$ \\
            \hline
            $\sin\alpha$ ($y$) & $+$ & $+$ & $-$ & $-$ \\
            \hline
            $\tan\alpha$ ($y/x$) & $+$ & $-$ & $+$ & $-$ \\
            \hline
            $\cot\alpha$ ($x/y$) & $+$ & $-$ & $+$ & $-$ \\
            \hline
        \end{tabular}
        \end{center}
        \item \textit{Điều kiện xác định và miền giá trị}:
        $$-1 \leq \sin\alpha \leq 1; \quad -1 \leq \cos\alpha \leq 1 \quad (\forall \alpha \in \mathbb{R}).$$
        $\tan\alpha$ xác định khi $\alpha \neq \dfrac{\pi}{2} + k\pi$ ($k \in \mathbb{Z}$); $\cot\alpha$ xác định khi $\alpha \neq k\pi$ ($k \in \mathbb{Z}$).
    \end{itemize}
    \item \textbf{c) Sản phẩm}:
    \begin{itemize}
        \item HS ghi chép định nghĩa đầy đủ và bảng xét dấu.

    \end{itemize}

        \item \textit{Thực hiện có giàn giáo}: Với HS trung bình - yếu, GV hướng dẫn thần chú trực quan: "Chiếu xuống $Ox$ được $\cos$, chiếu sang $Oy$ được $\sin$; cùng dấu thì $\tan, \cot$ dương, trái dấu thì $\tan, \cot$ âm".
        \item \textit{Báo cáo - Nhận định}: Gọi đại diện HS trả lời, chuẩn hóa bảng dấu.
    \end{itemize}
\end{itemize}

    \end{itemize}
    \item \textbf{c) Sản phẩm}: Học sinh ghi nhớ bảng giá trị, bấm MTCT chuẩn xác và nắm vững cấu trúc câu lệnh hỏi AI.

\subsubsection*{1. Hoạt động 1: Mở đầu (Khởi động - 6 phút)}

\paragraph{a) Mục tiêu:}
Từ định lý Pythagore trong tam giác vuông trên hệ tọa độ $Oxy$ và tính đối xứng của đường tròn lượng giác, khơi gợi nhu cầu tìm kiếm các công thức liên hệ chặt chẽ giữa các giá trị lượng giác.

\paragraph{b) Nội dung:}
\begin{itemize}
    \item Điểm $M(x; y)$ nằm trên đường tròn lượng giác $x^2 + y^2 = 1$. Thay $x = \cos\alpha, y = \sin\alpha$ vào phương trình, ta thu được đẳng thức nào?
    \item Quan sát điểm $M(\cos\alpha; \sin\alpha)$ và điểm $M'$ đối xứng với $M$ qua trục $Ox$, tọa độ của $M'$ liên hệ như thế nào với tọa độ của $M$? Góc lượng giác tương ứng của $M'$ là góc nào?
\end{itemize}

\paragraph{c) Sản phẩm:}
\begin{itemize}
    \item Đẳng thức thu được: $\cos^2\alpha + \sin^2\alpha = 1$.
    \item Điểm đối xứng qua $Ox$ có tọa độ $(\cos\alpha; -\sin\alpha)$, tương ứng với góc đối $-\alpha$. Từ đó nhận ra mối liên hệ: $\cos(-\alpha) = \cos\alpha$ và $\sin(-\alpha) = -\sin\alpha$.
\end{itemize}

\paragraph{d) Tổ chức thực hiện:}
GV đặt câu hỏi gợi mở, hướng dẫn học sinh quan sát hình vẽ trực quan trên bảng, kết nối trực tiếp vào các hệ thức cơ bản.

\subsubsection*{2. Hoạt động 2: Hình thành kiến thức (22 phút)}

\paragraph{Nội dung 1: Các hệ thức lượng giác cơ bản (11 phút)}
\begin{itemize}
    \item \textbf{a) Mục tiêu}: Nắm vững và biết cách chứng minh 4 hệ thức lượng giác cơ bản; biết cách tính các giá trị lượng giác còn lại khi biết một giá trị lượng giác và khoảng của góc $\alpha$.
    \item \textbf{b) Nội dung}:
    \begin{itemize}
        \item Bốn hệ thức lượng giác cơ bản:
        \begin{enumerate}
            \item $\sin^2\alpha + \cos^2\alpha = 1$ với mọi $\alpha \in \mathbb{R}$.
            \item $1 + \tan^2\alpha = \dfrac{1}{\cos^2\alpha}$ với $\alpha \neq \dfrac{\pi}{2} + k\pi$ ($k \in \mathbb{Z}$).
            \item $1 + \cot^2\alpha = \dfrac{1}{\sin^2\alpha}$ với $\alpha \neq k\pi$ ($k \in \mathbb{Z}$).
            \item $\tan\alpha \cdot \cot\alpha = 1$ với $\alpha \neq k\dfrac{\pi}{2}$ ($k \in \mathbb{Z}$).
        \end{enumerate}
        \item \textit{Ví dụ áp dụng}: Cho $\sin\alpha = \dfrac{3}{5}$ với $\dfrac{\pi}{2} < \alpha < \pi$. Tính $\cos\alpha, \tan\alpha, \cot\alpha$.
        \textit{Lời giải chi tiết}:
        + Áp dụng $\sin^2\alpha + \cos^2\alpha = 1 \implies \cos^2\alpha = 1 - \sin^2\alpha = 1 - \left(\dfrac{3}{5}\right)^2 = \dfrac{16}{25}$.
        + Do $\dfrac{\pi}{2} < \alpha < \pi$ (Góc phần tư thứ II) nên $\cos\alpha < 0$. Do đó:
        $$\cos\alpha = -\sqrt{\dfrac{16}{25}} = -\dfrac{4}{5}.$$
        + Từ đó suy ra:
        $$\tan\alpha = \dfrac{\sin\alpha}{\cos\alpha} = \dfrac{3/5}{-4/5} = -\dfrac{3}{4}; \quad \cot\alpha = \dfrac{1}{\tan\alpha} = -\dfrac{4}{3}.$$
    \end{itemize}
    \item \textbf{c) Sản phẩm}: Lời giải mẫu hoàn chỉnh của học sinh vào vở ghi.
    \item \textbf{d) Tổ chức thực hiện}:
    \begin{itemize}
        \item GV cho HS tự chứng minh công thức (2) bằng cách chia hai vế của (1) cho $\cos^2\alpha$.
        \item \textit{Giàn giáo sư phạm}: Nhắc nhở HS yếu luôn phải kiểm tra dấu của góc phần tư trước khi khai căn bậc hai (tránh sai lầm kinh điển: $\cos\alpha = \pm \dfrac{4}{5}$ mà không loại nghiệm).
    \end{itemize}
\end{itemize}

\paragraph{Nội dung 2: Giá trị lượng giác của các góc có liên quan đặc biệt (11 phút)}
\begin{itemize}
    \item \textbf{a) Mục tiêu}: Hiểu và áp dụng các công thức lượng giác của góc đối nhau, bù nhau, hơn kém nhau $\pi$, phụ nhau (ghi nhớ theo khẩu quyết sư phạm: \textit{"Cos đối, Sin bù, Phụ chéo, Hơn kém pi tan cot"}).
    \item \textbf{b) Nội dung}:
    \begin{enumerate}
        \item \textbf{Góc đối nhau} ($\alpha$ và $-\alpha$):
        $$\cos(-\alpha) = \cos\alpha; \quad \sin(-\alpha) = -\sin\alpha; \quad \tan(-\alpha) = -\tan\alpha; \quad \cot(-\alpha) = -\cot\alpha.$$
        \item \textbf{Góc bù nhau} ($\alpha$ và $\pi - \alpha$):
        $$\sin(\pi - \alpha) = \sin\alpha; \quad \cos(\pi - \alpha) = -\cos\alpha; \quad \tan(\pi - \alpha) = -\tan\alpha; \quad \cot(\pi - \alpha) = -\cot\alpha.$$
        \item \textbf{Góc hơn kém nhau $\pi$} ($\alpha$ và $\alpha + \pi$):
        $$\tan(\alpha + \pi) = \tan\alpha; \quad \cot(\alpha + \pi) = \cot\alpha; \quad \sin(\alpha + \pi) = -\sin\alpha; \quad \cos(\alpha + \pi) = -\cos\alpha.$$
        \item \textbf{Góc phụ nhau} ($\alpha$ và $\dfrac{\pi}{2} - \alpha$):
        $$\sin\left(\dfrac{\pi}{2} - \alpha\right) = \cos\alpha; \quad \cos\left(\dfrac{\pi}{2} - \alpha\right) = \sin\alpha; \quad \tan\left(\dfrac{\pi}{2} - \alpha\right) = \cot\alpha; \quad \cot\left(\dfrac{\pi}{2} - \alpha\right) = \tan\alpha.$$
    \end{enumerate}
    \item \textbf{c) Sản phẩm}:
    \begin{itemize}
        \item Rút gọn biểu thức $A = \cos(\pi - \alpha) + \sin\left(\dfrac{\pi}{2} - \alpha\right) - \tan(\pi + \alpha) \cdot \cot(-\alpha)$:
        $$A = (-\cos\alpha) + (\cos\alpha) - (\tan\alpha) \cdot (-\cot\alpha) = 0 - (-1) = 1.$$
    \end{itemize}
    \item \textbf{d) Tổ chức thực hiện}:
    \begin{itemize}
        \item GV hướng dẫn học sinh cách dùng tính đối xứng hình học trên đường tròn lượng giác để ghi nhớ công thức.
        \item Cho HS đọc đồng thanh câu ghi nhớ: \textit{"Cos đối, Sin bù, Phụ chéo, Khác pi Tan"}.
    \end{itemize}
\end{itemize}

\subsubsection*{3. Hoạt động 3: Luyện tập củng cố (12 phút)}

\paragraph{a) Mục tiêu:}
Vận dụng tổng hợp các hệ thức lượng giác cơ bản và công thức góc liên quan đặc biệt để rút gọn biểu thức và tính giá trị biểu thức.

\paragraph{b) Nội dung:}
Học sinh thực hiện Phiếu học tập số 3 theo nhóm 4 em:
\begin{itemize}
    \item \textbf{Bài 1}: Cho $\tan\alpha = -2$. Tính giá trị của biểu thức $P = \dfrac{2\sin\alpha - 3\cos\alpha}{4\sin\alpha + 5\cos\alpha}$.
    \item \textbf{Bài 2}: Rút gọn biểu thức lượng giác:
    $$B = \dfrac{\sin(\pi - x) \cdot \cos\left(\dfrac{\pi}{2} - x\right)}{\cos(\pi + x) \cdot \sin\left(\dfrac{\pi}{2} + x\right)} + \tan^2 x.$$
    \item \textbf{Bài 3}: Chứng minh đẳng thức: $\dfrac{\sin x + \cos x - 1}{\sin x - \cos x + 1} = \dfrac{\cos x}{1 + \sin x}$ (với các điều kiện xác định thỏa mãn).
\end{itemize}

\paragraph{c) Sản phẩm (Lời giải chi tiết):}
\begin{itemize}
    \item \textbf{Bài 1}:
    Vì $\tan\alpha = -2 \implies \cos\alpha \neq 0$. Chia cả tử và mẫu của $P$ cho $\cos\alpha$:
    $$P = \dfrac{2\dfrac{\sin\alpha}{\cos\alpha} - 3}{4\dfrac{\sin\alpha}{\cos\alpha} + 5} = \dfrac{2\tan\alpha - 3}{4\tan\alpha + 5} = \dfrac{2(-2) - 3}{4(-2) + 5} = \dfrac{-7}{-3} = \dfrac{7}{3}.$$
    \item \textbf{Bài 2}:
    Áp dụng các góc liên quan đặc biệt:
    + $\sin(\pi - x) = \sin x$;
    + $\cos\left(\dfrac{\pi}{2} - x\right) = \sin x$;
    + $\cos(\pi + x) = -\cos x$;
    + $\sin\left(\dfrac{\pi}{2} + x\right) = \sin\left(\dfrac{\pi}{2} - (-x)\right) = \cos(-x) = \cos x$.
    Do đó:
    $$B = \dfrac{\sin x \cdot \sin x}{(-\cos x) \cdot \cos x} + \tan^2 x = -\dfrac{\sin^2 x}{\cos^2 x} + \tan^2 x = -\tan^2 x + \tan^2 x = 0.$$
    \item \textbf{Bài 3}:
    Biến đổi vế trái:
    Ta có: $(\sin x + \cos x - 1)(1 + \sin x) = \sin x + \sin^2 x + \cos x + \sin x \cos x - 1 - \sin x$
    $$= (\sin^2 x - 1) + \cos x + \sin x \cos x = -\cos^2 x + \cos x + \sin x \cos x = \cos x (1 - \cos x + \sin x).$$
    Mặt khác, vế phải nhân với mẫu của vế trái:
    $\cos x (\sin x - \cos x + 1) = \cos x (1 - \cos x + \sin x)$.
    Vì hai tích chéo bằng nhau nên đẳng thức được chứng minh.
\end{itemize}

\paragraph{d) Tổ chức thực hiện:}
\begin{itemize}
    \item Các nhóm làm việc trong 7 phút, ghi kết quả lên bảng phụ nhóm.
    \item GV chọn 2 nhóm treo bảng phụ, gọi học sinh nhóm khác nhận xét và đối chiếu lời giải.
    \item GV tổng kết, đánh giá điểm cộng học tập cho các nhóm hoàn thành tốt.
\end{itemize}

\subsubsection*{4. Hoạt động 4: Vận dụng và Hướng dẫn tự học (5 phút)}

Tổng kết toàn bộ kiến thức của Bài 1; vận dụng giải quyết bài toán mô phỏng chuyển động cơ học trong STEM; hướng dẫn tự học và chuẩn bị bài mới.

\paragraph{b) Nội dung:}
\begin{itemize}
    \item \textbf{Nhiệm vụ STEM thực tiễn}: Thiết kế mô hình con lắc đơn chuyển động với góc lệch cực đại $\alpha_0 = 30^\circ$. Biết chiều dài dây treo $l = 1\text{ m}$.
    Viết biểu thức tính độ cao $z(\alpha)$ của vật nặng so với vị trí cân bằng theo góc lệch $\alpha$. Tính $z$ khi $\alpha = 0^\circ, 15^\circ, 30^\circ$.
    \item \textbf{Nhiệm vụ AI tự học ở nhà}:
    Học sinh sử dụng trợ lý AI (Gemini/ChatGPT) để tự ôn tập bằng câu lệnh:
    \textit{"Hãy tạo cho tôi 5 câu hỏi trắc nghiệm khách quan có 4 đáp án về giá trị lượng giác của các góc liên quan đặc biệt kèm lời giải chi tiết và mẹo tránh sai lầm."}
\end{itemize}

\paragraph{c) Sản phẩm:}
\begin{itemize}
    \item Biểu thức độ cao con lắc đơn: $z(\alpha) = l - l\cos\alpha = l(1 - \cos\alpha) = 1 - \cos\alpha\text{ (m)}$.
    + Khi $\alpha = 0^\circ \implies z(0) = 1 - \cos 0^\circ = 0\text{ m}$ (Vị trí thấp nhất).
    + Khi $\alpha = 15^\circ \implies z(15^\circ) = 1 - \cos 15^\circ \approx 1 - 0{,}9659 = 0{,}0341\text{ m} = 3{,}41\text{ cm}$.
    + Khi $\alpha = 30^\circ \implies z(30^\circ) = 1 - \cos 30^\circ = 1 - \dfrac{\sqrt{3}}{2} \approx 0{,}134\text{ m} = 13{,}4\text{ cm}$.
    \item Ảnh chụp màn hình kết quả câu lệnh hỏi AI và bài làm nộp trên Google Classroom / Zalo nhóm lớp.
\end{itemize}

\paragraph{d) Tổ chức thực hiện:}

\textbf{MỨC 2: THÔNG HIỂU (Có giàn giáo hỗ trợ)}
